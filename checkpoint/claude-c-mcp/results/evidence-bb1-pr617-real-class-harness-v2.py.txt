async def test_02_full_page_maps_every_real_record(real: Any, tmp_path: Path) -> None:
    with engine_api(tmp_path, real["reader"]) as (server, token_path):
        async with session_for(adapter_for(server, token_path)) as s:

    assert ok is False, page
    body = page["data"]["events"]
    by_type: dict[str, list[dict[str, Any]]] = {}
    for e in body["events"]:
        by_type.setdefault(e["event_type"], []).append(e)
    assert body["gaps_in_range"] == []                      # nessun record reale reso CORRUPT
    assert [w["reason_code"] for w in body["unrecorded_windows"]] == ["UNCLEAN_TERMINATION"]

    assert all(e["outcome"] == "PENDING" for e in body["events"] if e["phase"] == "ATTEMPT")
    assert by_type["API_REQUEST"][0]["severity"] == "TRACE"
    exc = by_type["TASK_EXCEPTION"][0]
    assert (exc["component"], exc["reason_code"]) == ("thread:worker-x", "KeyError")
    assert exc["data"]["password"] == "[REDACTED]"
    assert by_type["ORDER_MATCHED"][0]["exchange_ts"] == "2025-11-09T12:01:10.500Z"
    assert by_type["ORDER_MATCHED"][0]["ts_utc"] != by_type["ORDER_MATCHED"][0]["exchange_ts"]
    assert by_type["UNCLEAN_TERMINATION"][0]["session_id"] == "s_current"
    text = str(page)
    for secret in ("cref-secret-1", "cref-secret-2", "hunter2"):
        assert secret not in text, secret

async def test_07_every_mcp_read_is_recorded_in_the_real_register(real: Any, tmp_path: Path) -> None:
    with engine_api(tmp_path, real["reader"]) as (server, token_path):
        adapter = adapter_for(server, token_path)
        async with session_for(adapter) as s:
            await call(s, "list_blackbox_events", {"strategy": "roserpina"})
            rid = adapter.last_request_id
            import time
            deadline = time.monotonic() + 5
            mine: list[dict[str, Any]] = []
            while time.monotonic() < deadline:
                rows = real["reader"].query({"type": "CONTROL_API_REQUEST", "correlation_id": rid}, 0, 50)["records"]
                mine = rows
                if len(mine) >= 2:
                    break
                time.sleep(0.05)
            ok, page = await call(s, "list_blackbox_events", {"correlation_id": rid})
    assert [r["phase"] for r in mine] == ["ATTEMPT", "RESULT"]
    assert mine[1]["data"]["returned"] == 2 and mine[1]["data"]["status"] == 200
    assert ok is False, page
    seen = [(e["phase"], e["outcome"], e["data"]["returned"]) for e in page["data"]["events"]["events"]]
    assert seen[:2] == [("ATTEMPT", "PENDING", None), ("RESULT", "VERIFIED", 2)]


async def test_08_filters_and_pagination(real: Any, tmp_path: Path) -> None:
    with engine_api(tmp_path, real["reader"]) as (server, token_path):
        async with session_for(adapter_for(server, token_path)) as s:
            out = {}
            for name, args in {
                "table_selection": {"table_id": 3, "selection_id": 123456},
                "phase": {"phase": "ATTEMPT"},
                "severity": {"severity": "TRACE"},
                "causation": {"causation_id": real["attempt"]},
                "category": {"category": "CONFIG", "origin": "USER"},
                "type": {"event_type": "ORDER_MATCHED", "outcome": "VERIFIED"},
                "session": {"session_id": "s_clean"},
            }.items():
                out[name] = await call(s, "list_blackbox_events", args)
            seen, cursor = [], 0
            for _ in range(20):
                ok, page = await call(s, "list_blackbox_events", {"after_seq": cursor, "limit": 3})
                assert ok is False, page
                seen += [e["seq"] for e in page["data"]["events"]["events"]]
                cursor = page["data"]["events"]["next_after_seq"]
                if not cursor:
                    break
    counts = {k: (len([e for e in v[1]["data"]["events"]["events"] if e["event_type"] != "CONTROL_API_REQUEST"])
                  if v[0] is False else v) for k, v in out.items()}
    assert counts == {"table_selection": 2, "phase": 3, "severity": 1, "causation": 1, "category": 1,
                      "type": 1, "session": 2}, counts
    assert seen == sorted(set(seen)) and seen[0] == 1


def test_09_outage_gap_from_the_real_writer(tmp_path: Path) -> None:
    """Guasto reale: una scrittura non critica fallisce, al ripristino BB-1 scrive BLACKBOX_GAP + RECOVERED."""
    box = bb.BlackBox(str(tmp_path / "o.sqlite"), session_id="s_o", pid=7)
    box.start_session()
    real_insert = box._insert
    calls = {"n": 0}

    def failing(rec: dict[str, Any]) -> int:
        calls["n"] += 1
        if calls["n"] == 1:
            raise sqlite3.OperationalError("disk I/O error (simulato)")
        return real_insert(rec)

    box._insert = failing  # type: ignore[method-assign]
    assert box.record("API_REQUEST", source="CONTROL_API", phase="OBSERVATION", outcome="NOT_APPLICABLE") is None
    box._insert = real_insert  # type: ignore[method-assign]
    box.record("CONFIG_CHANGE", critical=True, source="USER", phase="RESULT", outcome="VERIFIED")
    reader = BlackBoxReader(box)
    page = make_providers(reader)["blackbox_events"]({"limit": 100})
    assert [w["reason_code"] for w in page["unrecorded_windows"]] == ["BLACKBOX_OUTAGE"]
    gap = [e for e in page["events"] if e["event_type"] == "BLACKBOX_GAP"][0]
    assert gap["outcome"] == "UNKNOWN" and gap["data"]["refs_count"] == 0 and "refs" not in gap["data"]
    box.close()

def test_15_a_read_after_shutdown_is_not_a_false_crash(tmp_path: Path) -> None:

    box = bb.BlackBox(path, session_id="s_a", pid=11)
